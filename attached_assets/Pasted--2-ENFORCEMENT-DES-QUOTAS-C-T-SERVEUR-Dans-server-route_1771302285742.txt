=== 2. ENFORCEMENT DES QUOTAS CÔTÉ SERVEUR ===

Dans server/routes.ts, modifier la route POST /api/admin/trips (création de voyage) :

Remplacer la logique de quota existante (qui utilise adminUser.maxTrips) par une logique basée sur le plan :
```ts
// Ancienne logique à remplacer :
if (adminUser.role !== "super_admin" && adminUser.maxTrips !== -1) {
  const [countResult] = await db.select({ count: sql<number>`count(*)::int` }).from(trips).where(eq(trips.userId, userId));
  if ((countResult?.count ?? 0) >= adminUser.maxTrips) {
    return res.status(403).json({ message: `Quota atteint (${adminUser.maxTrips} voyage(s) maximum)` });
  }
}

// Nouvelle logique :
if (adminUser.role !== "super_admin") {
  const plan = adminUser.plan || "free";
  const { PLAN_LIMITS } = await import("./stripe");
  const limits = PLAN_LIMITS[plan as keyof typeof PLAN_LIMITS] || PLAN_LIMITS.free;
  
  if (limits.maxTrips !== -1) {
    // Compter seulement les voyages NON archivés (draft + active)
    const [countResult] = await db
      .select({ count: sql<number>`count(*)::int` })
      .from(trips)
      .where(and(
        eq(trips.userId, userId),
        ne(trips.status, "archived")
      ));
    const currentCount = countResult?.count ?? 0;
    if (currentCount >= limits.maxTrips) {
      return res.status(403).json({ 
        message: `Quota atteint (${limits.maxTrips} voyage(s) actif(s) maximum sur le plan ${limits.name}). Archivez des voyages ou passez au plan supérieur.`,
        upgradeUrl: "/pricing"
      });
    }
  }
}
```
